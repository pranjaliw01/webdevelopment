        <button onclick="changeImage()">Change Image</button>

        <button onclick="changeBackgroundColor()">Change BackgroundColor</button>

    <script>
        function changeContent() {
            let title = document.getElementById("title");
            let paragraph = document.getElementById("paragraph");

            title.innerHTML = "Hello Student";
            paragraph.innerHTML = "Content has been changed!";
        }

        function changeImage() {
            let image = document.getElementById("image");

            image.src = "https://static.vecteezy.com/system/resources/thumbnails/057/068/323/small/single-fresh-red-strawberry-on-table-green-background-food-fruit-sweet-macro-juicy-plant-image-photo.jpg";
        }

        function changeLink() {
            let link = document.getElementById("link");

            link.href = "https://www.google.com";
            link.innerHTML = "Google";
        }

        function changeFontSize() {
            let text = document.getElementById("paragraph");

            text.style.fontSize = "30px";
        }

        function changeBackgroundColor() {
            let page = document.getElementById("box");

            page.style.backgroundColor = "lightpink";
        }
    </script>
